cmake_minimum_required(VERSION 3.20)
project(EMCICalcWorkflowTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
add_executable(workflow_tests workflow_tests.cpp ../../src/inputworkflows.cpp ../../src/programmerpanel.cpp ../../src/keypaddesigner.cpp ../../src/workflowtranslations.cpp ../../src/calculatorpanel.cpp ../../src/matheditor.cpp ../../src/calculatorthemes.cpp)
target_compile_definitions(workflow_tests PRIVATE EMCICALC_ANDROID_UI=1)
target_link_libraries(workflow_tests PRIVATE Qt6::Widgets)
target_include_directories(workflow_tests PRIVATE ../../src)
enable_testing()
add_test(NAME workflow_tests COMMAND workflow_tests)
set_tests_properties(workflow_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
