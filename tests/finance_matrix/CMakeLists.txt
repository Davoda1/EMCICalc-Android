cmake_minimum_required(VERSION 3.20)
project(FinanceInteractionRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "control or final source")
set(EMCICALC_ANDROID_UI ON CACHE BOOL "Test production Android presentation" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "Only the targeted regression harness" FORCE)
option(SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" ON)
if(SANITIZE)
 add_compile_options(-g -O0 -fno-omit-frame-pointer -fsanitize=address,undefined)
 add_link_options(-fsanitize=address,undefined)
endif()
add_compile_definitions(EMCICALC_ANDROID_UI=1)
find_package(Qt6 6.10.3 EXACT REQUIRED COMPONENTS Core Gui Widgets Network Test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(QALCULATE_PRIVATE REQUIRED IMPORTED_TARGET libxml-2.0 icu-uc)
add_subdirectory("${SOURCE_ROOT}" app EXCLUDE_FROM_ALL)
add_library(finance_scrolltrace SHARED scrolltrace.cpp)
target_link_libraries(finance_scrolltrace PRIVATE Qt6::Widgets ${CMAKE_DL_LIBS})
get_target_property(APP_SOURCES EMCICalc SOURCES)
set(PROBE_SOURCES)
foreach(SOURCE IN LISTS APP_SOURCES)
 if(SOURCE STREQUAL "src/main.cpp")
  continue()
 endif()
 if(IS_ABSOLUTE "${SOURCE}" OR SOURCE MATCHES "^\\$<")
  list(APPEND PROBE_SOURCES "${SOURCE}")
 else()
  list(APPEND PROBE_SOURCES "${SOURCE_ROOT}/${SOURCE}")
 endif()
endforeach()
get_target_property(APP_LIBRARIES EMCICalc LINK_LIBRARIES)
add_executable(finance_mainwindow_probe mainwindow_probe.cpp ${PROBE_SOURCES})
target_include_directories(finance_mainwindow_probe PRIVATE "${SOURCE_ROOT}/src")
target_link_libraries(finance_mainwindow_probe PRIVATE ${APP_LIBRARIES} Qt6::Test PkgConfig::QALCULATE_PRIVATE)
add_executable(finance_picker_probe picker_probe.cpp
 "${SOURCE_ROOT}/src/calculatorpanel.cpp" "${SOURCE_ROOT}/src/keypaddesigner.cpp")
target_include_directories(finance_picker_probe PRIVATE "${SOURCE_ROOT}/src")
target_link_libraries(finance_picker_probe PRIVATE emci_finance_ui emci_math_ui Qt6::Widgets Qt6::Test)
